import numpy as np
import pandas as pd

# 1. NUMPY
a = np.array([90, 100, 110, 120])          # 1-D array
m = np.array([[1, 2, 3], [4, 5, 6]])       # 2-D array

# Attributes
print("ndim:", a.ndim, m.ndim)             # rank: 1, 2
print("shape:", a.shape, m.shape)          # (4,) (2, 3)
print("size:", a.size, m.size)             # 4, 6
print("dtype:", a.dtype)                   # int64

# Other ways to create arrays

print("arange:", np.arange(0, 10, 2))      # stop excluded
print("linspace:", np.linspace(0, 1, 5))

# Indexing and slicing
print("a[0], a[-1]:", a[0], a[-1])
print("a[1:3]:", a[1:3])
print("m[1, 2]:", m[1, 2])
print("row 0:", m[0], " column 1:", m[:, 1])

# Operations (element-wise)

print("a * 2:", a * 2)
print("a + a:", a + a)

import pandas as pd
import numpy as np
from sklearn.metrics import mean_squared_error
from sklearn.linear_model import LinearRegression
from sklearn.model_selection import train_test_split

# Load data
# df = pd.read_csv('Salary_Data.csv', sep=',')
df = pd.DataFrame({
  'YearsExperience': [1.1, 1.3, 1.5, 2.0, 2.2, 2.9, 3.0, 3.2, 3.2, 3.7,
                      3.9, 4.0, 4.0, 4.1, 4.5, 4.9, 5.1, 5.3, 5.9, 6.0,
                      6.8, 7.1, 7.9, 8.2, 8.7, 9.0, 9.5, 9.6, 10.3, 10.5],
  'Salary': [39343, 46205, 37731, 43525, 39891, 56642, 60150, 54445, 64445, 57189,
             63218, 55794, 56957, 57081, 61111, 67938, 66029, 83088, 81363, 93940,
             91738, 98273, 101302, 113812, 109431, 105582, 116969, 112635, 122391, 121872]})

print(df.head())
print(df.shape)            # (30, 2)
print(df.isnull().sum())   # 0 missing in both columns

# Data preparation: 2-D column arrays
X = np.array(df['YearsExperience']).reshape(-1, 1)
Y = np.array(df['Salary']).reshape(-1, 1)
print(X.shape, Y.shape)    # (30, 1) (30, 1)

# Train-test split: 80% train, 20% test
X_train, x_test, Y_train, y_test = train_test_split(
    X, Y, test_size=0.2, shuffle=True, random_state=10)

# Fit the model
model = LinearRegression()
model.fit(X_train, Y_train)

# Score (R squared) on training and testing data
print("Train score:", model.score(X_train, Y_train))
print("Test score:", model.score(x_test, y_test))

# Mean squared error on test data
y_pred = model.predict(x_test)
mse = mean_squared_error(y_test, y_pred)
print("Mean squared error:", mse)
print("Root mean squared error:", np.sqrt(mse))

import tensorflow as tf
import numpy as np
import matplotlib.pyplot as plt

# Training data: inputs (features) and correct outputs (labels)
celsius    = np.array([-40, -10, 0, 8, 15, 22, 38], dtype=float)
fahrenheit = np.array([-40, 14, 32, 46, 59, 72, 100], dtype=float)

# Model: one Dense layer with one neuron
layer = tf.keras.layers.Dense(units=1, input_shape=[1])
model = tf.keras.Sequential([layer])

# Compile: how to measure error and how to reduce it
model.compile(loss='mean_squared_error',
              optimizer=tf.keras.optimizers.Adam(0.1))

# Train
history = model.fit(celsius, fahrenheit, epochs=500, verbose=False)
print("Finished training the model")

# Plot loss against epochs
plt.xlabel('Epoch Number')
plt.ylabel('Loss Magnitude')
plt.plot(history.history['loss'])
plt.show()

# Predict
print(model.predict(np.array([100.0])))
print("Formula answer:", 100 * 1.8 + 32)      # 212.0
print("Weight and bias:", layer.get_weights())
